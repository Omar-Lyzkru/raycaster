cmake_minimum_required(VERSION 3.16)
project(ConsoleRaycaster LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# The renderer uses Win32. The calculation tests also run on Linux and macOS.
if(WIN32)
    add_executable(raycaster "1st project/confpsbutbetter.cpp")
    target_link_libraries(raycaster PRIVATE user32)
    if(MSVC)
        target_compile_options(raycaster PRIVATE /W4)
    else()
        target_compile_options(raycaster PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(raycaster_tests tests/raycaster_tests.cpp)
    target_include_directories(raycaster_tests PRIVATE "1st project")
    if(MSVC)
        target_compile_options(raycaster_tests PRIVATE /W4)
    else()
        target_compile_options(raycaster_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    foreach(group coordinates bounds collision projection rays)
        add_test(NAME ${group} COMMAND raycaster_tests ${group})
    endforeach()
endif()
